extends SceneTree

# Temporary experiment: public actions and real world turns, no saves or scene startup.
var ranks: Dictionary = {}
var runs: int = 100
var mode: String = "baseline"
var rows: Array[Dictionary] = []
var boss: Dictionary = {}
var opening_rage: int = 0
var policy: String = "basic"
var defensive_cooldowns: bool = true


func _initialize() -> void:
	var config: Dictionary = JSON.parse_string(FileAccess.get_file_as_string("res://reports/sim_config.json"))
	mode = config.mode
	runs = config.runs
	opening_rage = int(config.get("rage", 0))
	policy = String(config.get("policy", "basic"))
	for id: String in SkillRank.all_ranks():
		ranks[id] = SkillRank.catalog(StringName(id))
	for arg in OS.get_cmdline_user_args():
		if arg.is_valid_int():
			runs = int(arg)
			continue
		if arg.begins_with("runs="):
			runs = int(arg.trim_prefix("runs="))
		else:
			mode = arg
	call_deferred("_run")


func _run() -> void:
	if mode == "efficient":
		for stage in ["mixed3", "mixed4"]:
			var gears: Array[String] = ["janos"]
			if stage == "mixed4":
				gears.append("dual")
			for gear in gears:
				for level in [2, 3]:
					_measure(stage, gear, level, 5, 1, false)
				for matchup in [[1, 3], [2, 4], [3, 5]]:
					_measure(stage, gear, matchup[0], matchup[1], 3, false)
		_save()
		return
	if mode == "full_kit":
		for stage in ["starter", "warrior", "mixed3", "mixed4"]:
			for level in [2, 3]:
				_measure(stage, "janos", level, 5, 1, false)
			for matchup in [[1, 3], [2, 4], [3, 5]]:
				_measure(stage, "janos", matchup[0], matchup[1], 3, false)
		_save()
		return
	if mode == "symmetric":
		_measure("starter", "janos", 1, 3, 1, false)
		for matchup in [[1, 3], [2, 4], [3, 5]]:
			_measure("mixed3", "janos", matchup[0], matchup[1], 3, false)
		opening_rage = 40
		for matchup in [[1, 3], [2, 4], [3, 5]]:
			_measure("mixed3", "janos", matchup[0], matchup[1], 3, false)
		opening_rage = 0
		boss = {"name": "boss8", "level": 8, "health": 280, "armor": 240,
			"minimum": 12, "maximum": 16, "interval": 2.5}
		_measure("mixed3", "janos", 5, 8, 1, true)
		_save()
		return
	if mode == "prepared":
		for matchup in [[1, 3], [2, 4], [3, 5]]:
			_measure("mixed3", "janos", matchup[0], matchup[1], 3, false)
		boss = {"name": "boss8", "level": 8, "health": 280, "armor": 240,
			"minimum": 12, "maximum": 16, "interval": 2.5}
		opening_rage = 0
		for level in [4, 5]:
			_measure("mixed3", "janos", level, 8, 1, true)
		_measure("warrior", "janos", 5, 8, 1, true)
		_save()
		return
	if mode == "confirm":
		_confirm()
		_save()
		return
	for stage in ["starter", "early", "mid", "warrior", "mixed3", "mixed4"]:
		for matchup in [[1, 1, 1], [1, 2, 1], [1, 3, 1], [1, 4, 1],
			[1, 1, 2], [1, 1, 3], [1, 3, 2], [1, 3, 3], [3, 5, 3],
			[2, 2, 2], [3, 3, 2], [4, 4, 2], [5, 5, 2], [2, 4, 1], [3, 5, 1], [2, 4, 3]]:
			_measure(stage, "janos", matchup[0], matchup[1], matchup[2], false)
	for matchup in [[1, 1, 1], [1, 3, 1], [1, 1, 2], [1, 1, 3]]:
		_measure("starter", "travel", matchup[0], matchup[1], matchup[2], false)
	for level in [3, 5, 8, 10, 20]:
		for stage in ["warrior", "mixed3", "mixed4"]:
			_measure(stage, "janos", level, 20, 1, true)
	for profile in [
		{"name": "boss7", "level": 7, "health": 220, "armor": 180, "minimum": 10, "maximum": 14, "interval": 2.5},
		{"name": "boss8", "level": 8, "health": 280, "armor": 240, "minimum": 12, "maximum": 16, "interval": 2.5},
		{"name": "boss8_heavy", "level": 8, "health": 350, "armor": 300, "minimum": 14, "maximum": 18, "interval": 2.5},
	]:
		boss = profile
		for level in [3, 4, 5]:
			for stage in ["starter", "warrior", "mixed3", "mixed4"]:
				_measure(stage, "janos", level, int(boss.level), 1, true)
	_save()


func _save() -> void:
	var file := FileAccess.open("res://reports/balance_%s.json" % mode, FileAccess.WRITE)
	file.store_string(JSON.stringify(rows, "\t"))
	print("Saved %d scenarios to reports/balance_%s.json" % [rows.size(), mode])
	quit()


func _confirm() -> void:
	for matchup in [["starter", 1, 3, 1], ["starter", 1, 4, 1],
		["starter", 1, 1, 2], ["starter", 1, 1, 3], ["early", 1, 3, 1],
		["mid", 1, 3, 2], ["warrior", 1, 3, 3], ["mixed3", 1, 3, 3],
		["mixed3", 2, 4, 3], ["mixed3", 3, 5, 3], ["mixed4", 3, 5, 3]]:
		_measure(matchup[0], "janos", matchup[1], matchup[2], matchup[3], false)
	_measure("starter", "travel", 1, 1, 2, false)
	_measure("starter", "travel", 1, 3, 1, false)
	boss = {"name": "boss8", "level": 8, "health": 280, "armor": 240,
		"minimum": 12, "maximum": 16, "interval": 2.5}
	for level in [4, 5]:
		_measure("mixed3", "janos", level, 8, 1, true)
	_measure("warrior", "janos", 5, 8, 1, true)


func _measure(stage: String, gear: String, level: int, enemy_level: int,
	count: int, stalker: bool) -> void:
	var wins := 0
	var health: Array[float] = []
	var turns: Array[int] = []
	var unfinished := 0
	for seed_value in range(runs):
		var result := _battle(stage, gear, level, enemy_level, count, stalker, seed_value + 1000)
		if result.win:
			wins += 1
			health.append(result.health)
			turns.append(result.turns)
		if result.unfinished:
			unfinished += 1
	health.sort()
	turns.sort()
	var row := {"stage": stage, "gear": gear, "hero_level": level,
		"policy": policy,
		"opening_rage": opening_rage,
		"enemy_level": enemy_level, "count": count, "stalker": stalker,
		"runs": runs, "win_percent": 100.0 * wins / runs,
		"median_win_hp_percent": 0.0 if health.is_empty() else health[health.size() / 2],
		"median_win_turns": 0 if turns.is_empty() else turns[turns.size() / 2],
		"unfinished": unfinished}
	row["boss_profile"] = boss.get("name", "current") if stalker else ""
	rows.append(row)
	print(JSON.stringify(row))


func _battle(stage: String, gear: String, level: int, enemy_level: int,
	count: int, stalker: bool, seed_value: int) -> Dictionary:
	var world := GridWorld.new(Rect2i(0, 0, 9, 9), Vector2i(4, 4), seed_value, false)
	world.hero.level = level
	if gear in ["janos", "dual", "two_hand"]:
		StarterGear.equip_outfit(world.hero)
	for skill: SkillRank in ranks.values():
		var threshold := 0 if stage == "starter" else (100 if stage == "early" else 240)
		if stage in ["warrior", "mixed3", "mixed4", "warrior_shaman", "warrior_paladin", "warrior_rogue"]:
			threshold = 10000
		var categories: Array[StringName] = [&"Warrior"]
		if stage in ["mixed3", "mixed4"]:
			categories.append_array([&"Shaman", &"Paladin"])
		if stage == "mixed4":
			categories.append(&"Rogue")
		if stage == "warrior_shaman":
			categories.append(&"Shaman")
		if stage == "warrior_paladin":
			categories.append(&"Paladin")
		if stage == "warrior_rogue":
			categories.append(&"Rogue")
		if skill.class_tab in categories and skill.progress_required <= threshold:
			# Already-learned setup: refresh once after assigning the retained catalogue.
			if world.hero.find_skill(skill.id) == null and world.hero.level >= skill.training_level:
				world.hero.learned_skills.append(skill)
	if gear == "dual":
		# Janos sells this one-handed Dirk; keep the rest of his free outfit.
		world.hero.inventory[0] = ItemData.instance(2139)
		assert(InventoryRules.equip(world.hero, 0, "off_hand"))
	elif gear == "two_hand":
		# Janos's level-one Short Staff removes the shield through the normal equip action.
		world.hero.inventory[0] = ItemData.instance(2132)
		assert(InventoryRules.equip(world.hero, 0, "main_hand"))
		assert(not world.hero.equipment.has("off_hand"))
	world.hero.refresh_stats()
	world.hero.health = world.hero.max_health
	world.hero.rage = opening_rage
	var health_tracking := {"last": world.hero.health, "damage": 0, "healing": 0}
	var health_listener := func(_message: String) -> void:
		var difference: int = world.hero.health - int(health_tracking.last)
		if difference < 0:
			health_tracking.damage -= difference
		elif difference > 0:
			health_tracking.healing += difference
		health_tracking.last = world.hero.health
	world.message_added.connect(health_listener)
	for index in range(count):
		var offset: Vector2i = [Vector2i.UP, Vector2i(-1, -1), Vector2i(1, -1), Vector2i.RIGHT][index]
		var enemy := GridActor.new(world.player_tile + offset)
		var identity: int = [299, 69, 80, 38, 103][clampi(enemy_level - 1, 0, 4)]
		var data: Dictionary = NorthshireData.CREATURES[str(identity)]
		enemy.title = "Trial enemy"
		enemy.stalker = stalker
		enemy.spawn_id = "trial/%d" % index
		enemy.awards_experience = false
		enemy.relationship = GridActor.Relationship.HOSTILE
		enemy.engaged = true
		enemy.max_health = int(data.health) if not stalker else 494
		enemy.health = enemy.max_health
		enemy.melee.level = enemy_level
		enemy.melee.armor = int(data.armor) if not stalker else 888
		enemy.melee.damage_min = float(data.minimum) if not stalker else 31.0
		enemy.melee.damage_max = float(data.maximum) if not stalker else 38.0
		enemy.melee.attack_power = int(data.power) if not stalker else 16
		enemy.melee.interval = float(data.interval)
		if stalker and not boss.is_empty():
			enemy.max_health = int(boss.health)
			enemy.health = enemy.max_health
			enemy.melee.armor = int(boss.armor)
			enemy.melee.damage_min = float(boss.minimum)
			enemy.melee.damage_max = float(boss.maximum)
			enemy.melee.attack_power = 0
			enemy.melee.interval = float(boss.interval)
		world.actors.append(enemy)
	var enemies := world.actors.duplicate()
	while not world.is_terminal() and world.turn_count < 300:
		var living: Array[GridActor] = []
		for enemy: GridActor in enemies:
			if enemy.alive:
				living.append(enemy)
		if living.is_empty():
			break
		living.sort_custom(func(a: GridActor, b: GridActor) -> bool: return a.health < b.health)
		var target := living[0]
		if world.pending_skill != null:
			world.continue_cast()
			continue
		if world.pending_melee != null:
			world.continue_melee()
			continue
		# Engage/retarget through the same committed bump action as a human player.
		if policy == "basic" and world.auto_attack_target != target and GridWorld.tile_distance(world.player_tile, target.tile) == 1:
			world.request_melee(target)
			continue
		var acted := false
		var choices: Array[String] = []
		if float(world.hero.health) / world.hero.max_health < 0.6:
			choices.append_array(["divine_shield_1", "evasion_1", "feral_spirit_1"])
		choices.append("battle_shout_1")
		if living.size() > 1:
			choices.append("whirlwind_1")
		choices.append_array(["execute_1", "earth_shock_1", "lightning_shield_1",
			"windfury_weapon_1", "strength_of_earth_totem_1", "blessing_of_might_1",
			"seal_of_command_1", "instant_poison_1", "slice_and_dice_1", "feral_spirit_1",
			"judgement_1", "mortal_strike_1"])
		if world.hero.combo_points >= 4:
			choices.append("eviscerate_1")
		# Reserve Heroic Strike only with surplus rage, preserving active attacks.
		if world.hero.rage >= 65:
			choices.append("heroic_strike_1")
		if policy == "efficient":
			choices = _efficient_choices(world, living.size(), target)
		for id in choices:
			if not defensive_cooldowns and id in ["evasion_1", "divine_shield_1"]:
				continue
			var skill := world.hero.find_skill(StringName(id))
			if skill == null or skill.cost(world.hero) > world.hero.rage:
				continue
			if policy == "efficient" and world.auto_attack_target != target:
				if skill.target != SkillRank.Target.ENEMY:
					continue
			if world.hero.buffs.has(skill.id) and skill.effect == SkillRank.Effect.ARMOR:
				continue
			if skill.effect == SkillRank.Effect.SUMMON and world.actors.any(
				func(a: GridActor) -> bool: return a.alive and a.summon_skill == skill.id):
				continue
			if skill.family == &"feral_spirit" and world.actors.any(
				func(a: GridActor) -> bool: return a.alive and a.summon_kind == "wolf"):
				continue
			var skill_target: GridActor = target if skill.target == SkillRank.Target.ENEMY else null
			if world.cast_skill(skill.id, skill_target):
				acted = true
				break
		if not acted:
			if world.auto_attack_target != target and GridWorld.tile_distance(world.player_tile, target.tile) == 1:
				world.request_melee(target)
			elif GridWorld.tile_distance(world.player_tile, target.tile) > 1:
				if not world.move_player((target.tile - world.player_tile).sign()):
					world.wait_turn()
			else:
				world.wait_turn()
	var won: bool = enemies.all(func(a: GridActor) -> bool: return not a.alive) and not world.is_player_dead()
	var result := {"win": won, "health": 100.0 * world.hero.health / world.hero.max_health,
		"health_hp": world.hero.health, "max_health_hp": world.hero.max_health,
		"damage_hp": health_tracking.damage, "healing_hp": health_tracking.healing,
		"turns": world.turn_count, "unfinished": not won and not world.is_terminal()}
	world.message_added.disconnect(health_listener)
	world.discard()
	return result


func _efficient_choices(world: GridWorld, count: int, target: GridActor) -> Array[String]:
	var choices: Array[String] = []
	var hero := world.hero
	var fraction := float(hero.health) / hero.max_health
	var protected := hero.buffs.has(&"divine_shield_1") or hero.buffs.has(&"evasion_1")
	if not protected and fraction < (0.85 if count > 1 else 0.5):
		choices.append("evasion_1")
		choices.append("divine_shield_1")
	choices.append("execute_1")
	if hero.combo_points >= 3:
		choices.append("eviscerate_1")
	if fraction < 0.6 and (count > 1 or target.max_health > 100):
		choices.append("feral_spirit_1")
	if count > 1 or target.health > 20:
		choices.append("lightning_shield_1")
	choices.append("battle_shout_1")
	if count > 1 or target.health > 30:
		choices.append("instant_poison_1")
	if count > 1:
		choices.append("whirlwind_1")
	choices.append("earth_shock_1")
	if count == 1 and target.max_health > 100:
		choices.append("feral_spirit_1")
	if target.health > 30 or count > 1:
		choices.append_array(["windfury_weapon_1", "strength_of_earth_totem_1",
			"blessing_of_might_1", "seal_of_command_1"])
		if hero.combo_points >= 3:
			choices.append("slice_and_dice_1")
	choices.append_array(["judgement_1", "mortal_strike_1"])
	if hero.rage >= 65:
		choices.append("heroic_strike_1")
	return choices
